% Acceptance occurs during the exchange; incorporation occurs only in later,
% independently scheduled local commits.
\begin{tikzpicture}[
  participant/.style={draw=swblue, thick, rounded corners=3pt, fill=swlightblue,
                      minimum width=34mm, minimum height=8mm,
                      font=\small\bfseries, align=center},
  lifeline/.style={draw=swgray!65, densely dotted},
  msg/.style={-{Latex[length=2.2mm]}, thick, draw=swgray},
  return/.style={msg, dashed},
  local/.style={draw=swgreen!80!black, thick, rounded corners=2pt,
                fill=swlightgreen, minimum width=40mm, minimum height=8mm,
                font=\footnotesize, align=center},
  note/.style={font=\scriptsize, text=swgray, align=center},
]
  \node[participant] (a) at (-4.0,0) {Initiator journal A};
  \node[participant] (b) at (4.0,0) {Acceptor journal B};
  \draw[lifeline] (a.south) -- (-4.0,-6.9);
  \draw[lifeline] (b.south) -- (4.0,-6.9);

  \draw[msg] (-4.0,-1.0) -- node[above, font=\footnotesize]
    {root-signed head A + control-path evidence} (4.0,-1.0);
  \node[local] (stage-a) at (4.0,-1.8) {verify and stage A\\status: accepted, not incorporated};
  \draw[return] (4.0,-2.7) -- node[above, font=\footnotesize]
    {root-signed head B + transport acknowledgment (accepted-index)} (-4.0,-2.7);
  \node[local] (stage-b) at (-4.0,-3.5) {verify and stage B\\status: accepted, not incorporated};

  \draw[swgray!55, dashed] (-5.8,-4.25) -- (5.8,-4.25);
  \node[note, fill=white, inner sep=2pt] at (0,-4.25)
    {later, on each journal's independent commit cadence};

  \node[local] (commit-a) at (-4.0,-5.15) {local commit includes head B};
  \node[local] (commit-b) at (4.0,-5.15) {local commit includes head A};
  \node[note] at (-4.0,-6.0) {later A head + inclusion proof\\proves A incorporated B};
  \node[note] at (4.0,-6.0) {later B head + inclusion proof\\proves B incorporated A};
\end{tikzpicture}
