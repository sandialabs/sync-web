\begin{tikzpicture}[
  field/.style={draw=swgray, thin, fill=white, minimum height=7mm,
                minimum width=20mm, align=center, inner xsep=3pt,
                font=\footnotesize\ttfamily},
  namespace/.style={field, draw=swblue, thick, fill=swlightblue,
                    minimum width=29mm},
  state/.style={field, draw=swgreen, fill=swlightgreen, minimum width=20mm},
  history/.style={field, fill=swlightblue, minimum width=18mm},
  document/.style={draw=swgreen, thick, rounded corners=2pt, fill=white,
                   minimum width=18mm, minimum height=8mm,
                   align=center, font=\scriptsize},
  container/.style={draw=swblue!55, rounded corners=4pt, dashed, inner sep=6pt},
  link/.style={-{Latex[length=2.2mm]}, thick, draw=swgray},
  recursive/.style={link, dashed, draw=swblue},
  annotation/.style={font=\scriptsize, text=swgray, align=center},
]
  % Indexed local history.
  \matrix (chain) [matrix of nodes, nodes={history}, column sep=-\pgflinewidth,
                   label={[font=\small\bfseries]above:Local permanent history}] at (0,0) {
    $\cdots$ & $n-2$ & $n-1$ & $n$ \\
  };

  % Reserved namespaces in the selected step.
  \node[namespace] (ns-state) at (-5.2,-2.0) {\texttt{*state*}};
  \node[namespace] (ns-transition) at (-5.2,-4.7) {\texttt{*transition*}};
  \node[namespace] (ns-bridge) at (-5.2,-6.7) {\texttt{*bridge*}};
  \node[namespace] (ns-crypto) at (-5.2,-9.0) {\texttt{*crypto*}};
  \node[container, fit=(ns-state)(ns-transition)(ns-bridge)(ns-crypto),
        label={[font=\small\bfseries]above:Step $n$ tree}] (step) {};
  \draw[link] (chain-1-4.south) -- ++(0,-4mm) -| (step.north);

  % State tree entries point to documents.
  \matrix (state-tree) [matrix of nodes, nodes={state}, column sep=-\pgflinewidth,
                        label={[font=\small\bfseries]above:State tree}] at (0.8,-2.0) {
    alice & report & \texttt{*time*} & $\cdots$ \\
  };
  \node[document, below=6mm of state-tree-1-1] (doc-a) {Document A};
  \node[document, below=6mm of state-tree-1-2] (doc-b) {Document B};
  \node[sw metadata, below=6mm of state-tree-1-3, text width=1.7cm,
        minimum height=8mm, font=\scriptsize] (time) {UTC time};
  \draw[link] (ns-state.east) -- (state-tree.west);
  \draw[link] (state-tree-1-1.south) -- (doc-a.north);
  \draw[link] (state-tree-1-2.south) -- (doc-b.north);
  \draw[link] (state-tree-1-3.south) -- (time.north);

  % Flat transition record.
  \matrix (transition) [matrix of nodes, nodes={field, minimum width=28mm},
                        column sep=-\pgflinewidth,
                        label={[font=\small\bfseries]above:Transition record}] at (-0.5,-4.7) {
    previous & operation \\
  };
  \draw[link] (ns-transition.east) -- (transition.west);

  % Bridge entry points to the retained head of another history.
  \matrix (bridge) [matrix of nodes, nodes={field, minimum width=25mm},
                    column sep=-\pgflinewidth,
                    label={[font=\small\bfseries]above:Bridge entry}] at (-0.5,-6.7) {
    peer info & remote head \\
  };
  \draw[link] (ns-bridge.east) -- (bridge.west);

  \matrix (remote) [matrix of nodes, nodes={history, minimum width=15mm},
                    column sep=-\pgflinewidth,
                    label={[font=\scriptsize\bfseries]above:Remote history}] at (4.4,-6.7) {
    $\cdots$ & $m-1$ & $m$ \\
  };
  \draw[link] (bridge-1-2.east) -- (remote.west);
  \node[sw primary, below=7mm of remote, text width=3.7cm, font=\scriptsize] (remote-step)
    {step $m$ tree with its own \texttt{*bridge*}};
  \node[font=\Large, text=swblue, right=7mm of remote-step] (more) {$\cdots$};
  \draw[link] (remote-1-3.south) -- (remote-step.north);
  \draw[recursive] (remote-step.east) -- node[above,annotation] {recurse} (more.west);

  % Flat cryptographic record.
  \matrix (crypto) [matrix of nodes, nodes={field, minimum width=28mm},
                    column sep=-\pgflinewidth,
                    label={[font=\small\bfseries]above:Cryptographic record}] at (-0.5,-9.0) {
    public-key & signature \\
  };
  \draw[link] (ns-crypto.east) -- (crypto.west);
\end{tikzpicture}
